\section{Problem \RMN{1}}

\subsection*{(a)}

The Hamiltonian is $H = -\frac{1}{2} \nabla^2 - \frac{1}{r}$,
construct wave functions with $4$ bases,
%
\begin{equation}
    \psi(\symbf{r}) = \sum_{i} c_i \phi_i = c_1 e^{-13.0 r^2} +
    c_2 e^{-1.96 r^2} + c_3 e^{-0.44 r^2} + c_4 e^{-0.12 r^2}.
\end{equation}
%
where $c_i$ are real numbers.

Define functional
%
\begin{equation}
    E[\psi] = \frac{\braket{\psi| H | \psi}}{\braket{\psi|\psi}},
\end{equation}
%
according to variational principle, the ground state energy $E$ satisfies
%
\begin{equation}
    \det (H_{ij} - S_{ij} E) = 0,
\end{equation}
%
where $H_{ij} = \braket{\phi_i | H | \phi_j}$, $S_{ij} = \braket{\phi_i | \phi_j}$.
Under spherical coordinates,
%
\begin{equation}
    \begin{aligned}
        \nabla^2 & = \frac{1}{r} \frac{\partial^2}{\partial r^2} r - \frac{\symbf{l}^2}{\hbar^2 r^2} \\
                 & = \frac{1}{r} \frac{\partial^2}{\partial r^2}r +
        \frac{1}{r^{2}\sin \theta} \frac{\partial}{\partial \theta}
        \Bigl(\sin \theta{\frac{\partial}{\partial \theta}}\Bigr)+
        \frac{1}{r^{2}\sin^{2}\theta} \frac{\partial^{2}}{\partial \varphi^{2}} \,,
    \end{aligned}
\end{equation}
%
here every $\phi(r)$ does not contain $\theta$ and $\varphi$, so the Laplacian reduces to
%
\begin{equation}
    \nabla^2 = \frac{1}{r} \frac{\partial^2}{\partial r^2} r.
\end{equation}
%
So
%
\begin{equation}
    \begin{aligned}
        \braket{\phi_i | H | \phi_j} & = \int \phi_i^\ast H \phi_j \, d\symbf{r}                              \\
                                     & = 4\pi\int_{0}^{\infty} \frac{-1 + 3 \alpha_j r - 2 \alpha_j^2 r^3}{r}
        e^{-(\alpha_i + \alpha_j)r^2} r^2\, dr.
    \end{aligned}
\end{equation}
%
Therefore
%
\begin{equation}
    H = \begin{pmatrix}
        0.577368  & 0.0717194 & -0.323206 & -0.437105 \\
        0.0717194 & 0.506469  & -1.00354  & -2.39107  \\
        -0.323206 & -1.00354  & -2.68809  & -7.46151  \\
        -0.437105 & -2.39107  & -7.46151  & -17.6552
    \end{pmatrix},
\end{equation}
%
and
%
\begin{equation}
    S = \begin{pmatrix}
        0.0420015 & 0.0962338 & 0.113012 & 0.117172 \\
        0.0962338 & 0.717457  & 1.49764  & 1.85622  \\
        0.113012  & 1.49764   & 6.74529  & 13.2875  \\
        0.117172  & 1.85622   & 13.2875  & 47.3596
    \end{pmatrix}.
\end{equation}

Then solve for different set of eigenvalues, we derive Table~\ref{tab:1-a}.

\section{Problem \RMN{2}}

\subsection*{(a)}

Set Hamiltonian to be $H = -\frac{1}{2}\nabla_1^2 - \frac{2}{r_1} - \frac{1}{2}\nabla_2^2 - \frac{2}{r_2}$, then
the Schrödinger equation is turned into $2$ single-electron equations, i.e.,
%
\begin{equation}
    \Bigl(-\frac{1}{2}\nabla^2 - \frac{2}{r}\Bigr) \psi(\symbf{r}) = E \psi(\symbf{r}).
\end{equation}

Construct single-electron wave function with $4$ bases,
%
\begin{equation}
    \psi(\symbf{r}) = \sum_{i} c_i \phi_i = c_1 e^{- 0.298073 r^2} +
    c_2 e^{- 1.242567 r^2} + c_3 e^{- 5.782948 r^2} + c_4 e^{-38.474970 r^2}.
\end{equation}
%
where $c_i, i=1,2,3,4$, are real numbers.
Now the ground state energy is \qty{-1.99426616}{\electronvolt}. The density is
%
\begin{equation}\label{eq:density}
    \lvert \psi(\symbf{r}) \rvert^2 = \psi^\ast(\symbf{r}) \psi(\symbf{r}) =
    \sum_{i, j} c_i^\ast c_j \phi_i^\ast(\symbf{r}) \phi_j(\symbf{r}),
\end{equation}
%
where $i$, $j = 1$, $2$, $3$, $4$. The radial probability distribution is
%
\begin{equation}
    \lvert \chi(\symbf{r}) \rvert^2 = r^2\int \lvert \psi(\symbf{r}) \rvert^2
    \, d\Omega = 4\pi r^2 \lvert \psi(\symbf{r}) \rvert^2.
\end{equation}
%
And, the exact value of single-electron wave function is
%
\begin{equation}
    \psi_{\text{exact}}(\symbf{r}) = \sqrt{\frac{8}{\pi}} e^{-2 r}, % This function is normalized.
\end{equation}
%
since the total wave function is $\Psi_{\text{exact}}(\symbf{r}_1, \symbf{r}_2) = \frac{8}{\pi} e^{-2(r_1 + r_2)}$.
See Griffith page $187$\footnotemark.
Then we can plot the density and radial probability distribution as a function of $r$
as Figure~\ref{fig:2-a:mini:subfig:a} and Figure~\ref{fig:2-a:mini:subfig:b}, respectively.

\footnotetext{Here Griffith only considered Hartree method but not Hartree--Fock method, so
    we don't need to construct an antisymmetric all-electron wave function.}

\subsection*{(b)}

Now consider the single-electron Schrödinger equation of $\psi_1(\symbf{r}_1)$ will be
%
\begin{equation}
    \Bigl(
    -\frac{1}{2}\nabla_1^2 - \frac{2}{r_1} + \int
    \frac{\psi_2^\ast(\symbf{r}_2)\psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert} d\symbf{r}_2
    \Bigr)
    \psi_1(\symbf{r}_1) = E \psi_1 (\symbf{r}_1),
\end{equation}
%
with
%
\begin{multline}
    \psi_2(\symbf{r}_2) = 0.08704173 e^{-0.298073 r^2} + 0.52509737 e^{-1.242567 r^2}\\
    + 0.51920929 e^{-5.782948 r^2} + 0.31233737 e^{-38.474970 r^2},
\end{multline}
%
as trial function (last question result), and
%
\begin{equation}
    \begin{aligned}
        \int \frac{\psi_2^\ast(\symbf{r}_2)\psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert} d\symbf{r}_2
         & = \sum_{i, j} c_i^\ast c_j \int
        \frac{\phi_i^\ast(r_2) \phi_j(r_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert} \, d\symbf{r}_2    \\
         & = \sum_{i, j} 2\pi c_i^\ast c_j \scaleint{7ex}_{\bs 0}^{\infty} \scaleint{7ex}_{\bs 0}^{\pi}
        \frac{\phi_i^\ast(r_2) \phi_j(r_2)}{\sqrt{r_1^2 + r_2^2 - 2 r_1 r_2 \cos\theta}}
        r_2^2 \sin \theta\, d\theta\, dr_2.
    \end{aligned}
\end{equation}

Left multiplied by $\bra{\phi_i(\symbf{r}_1)}$ and integrate, we have
%
\begin{equation}
    \begin{aligned}
        \Braket{\phi_i(\symbf{r}_1) | - \frac{1}{2}\nabla_1^2 - \frac{2}{r_1} | \psi_1(\symbf{r}_1)} +
        \Braket{\phi_i(\symbf{r}_1) |
            \int \frac{\psi_2^\ast(\symbf{r}_2)\psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert} d\symbf{r}_2 |
            \psi_1(\symbf{r}_1)}
        = E \braket{\phi_i(\symbf{r}_1) | \psi_1(\symbf{r}_1)}.
    \end{aligned}
\end{equation}
%
The eigenvalue problem will be
%
\begin{equation}
    \det (H_{ij} - S_{ij} E + K_{ij}) = 0,
\end{equation}
%
here $H_{ij} = \braket{\phi_i | - \frac{1}{2}\nabla_1^2 - \frac{2}{r_1} | \phi_j}$,
$S_{ij} = \braket{\phi_i | \phi_j}$, and
%
\begin{equation}
    K_{ij} = \sum_{m, n} 8\pi^2 c_m^\ast c_n
    \scaleint{7ex}_{\bs 0}^{\infty} r_1^2 \, dr_1
    \scaleint{7ex}_{\bs 0}^{\infty} r_2^2 \, dr_2
    \scaleint{7ex}_{\bs 0}^{\pi}
    \frac{\phi_i^\ast(r_1) \phi_m^\ast(r_2) \phi_n(r_2)\phi_j(r_1)}{\sqrt{r_1^2 + r_2^2 - 2 r_1 r_2 \cos\theta}}
    \sin\theta\, d\theta.
\end{equation}

Derive new set of $\{c_1, c_2, c_3, c_4\}$,
let $\psi_1(\symbf{r}_1) = \sum_{i=1}^{4} c_i \phi_i(\symbf{r}_1)$, and substitute into
%
\begin{equation}
    \Bigl(
    -\frac{1}{2}\nabla_2^2 - \frac{2}{r_2} +
    \int \frac{\psi_1^\ast(\symbf{r}_1)\psi_1(\symbf{r}_1)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert} d\symbf{r}_1
    \Bigr)
    \psi_2(\symbf{r}_2) = E \psi_2 (\symbf{r}_2),
\end{equation}
%
repeat this step until $E$ converges. The total energy is
%
\begin{equation}
    E_{\text{total}} = 2E -
    \int \frac{\psi_1^\ast(\symbf{r}_1)\psi_2^\ast(\symbf{r}_2) \psi_1(\symbf{r}_1)
        \psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert}
    \, d\symbf{r}_2 \, d\symbf{r}_1,
\end{equation}
%
Here $2E$ is because $\psi_1 = \psi_2$. See Figure~\ref{fig:2-b:mini:subfig:a} for reference.

\section{Problem \RMN{3}}

\subsection*{(a)}

By neglecting the electron-electron interactions, we have
%
\begin{equation}
    H = -\frac{1}{2}\nabla_1^2 - \frac{1}{2}\nabla_2^2
    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_1 \rvert} - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_2 \rvert}
    - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_1 \rvert} - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_2 \rvert}
    + \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert},
\end{equation}
%
The single-electron wave function is
%
\begin{equation}
    \psi(\symbf{r}) = \sum_{i} \sum_n c_{i, m}\phi_{i}^{R_m}(r) =
    \sum_{i=1}^{4}\sum_{m=1}^{2} c_{i, m} e^{-\alpha_i (r - R_m)^2}.
\end{equation}
%
So corresponding Schrödinger equation of $\psi_1(\symbf{r}_1)$ is
%
\begin{equation}
    \Bigl(- \frac{1}{2}\nabla_1^2
    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_1 \rvert}
    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_2 \rvert}
    + \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert}
    \Bigr)
    \psi_1(\symbf{r}_1) = E \psi_1 (\symbf{r}_1).
\end{equation}
%
Left multiplied by $\Bigl( \phi_i^{R_m} \Bigr)^\ast(\symbf{r}_1)$ and integrate, we have
%
\begin{equation}
    \Braket{\phi_i^{R_m} |
        - \frac{1}{2}\nabla_1^2
        - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_1 \rvert}
        - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_2 \rvert}
        + \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert} | \psi_1}
    = E \Braket{\phi_i^{R_m} | \psi_1}.
\end{equation}

So the eigenvalue problem will be
%
\begin{equation}
    \det(T_{ij, mn} + N^E_{ij, mn} + N^N_{ij, mn} - S_{ij, mn} E) = 0.
\end{equation}
%
where $T_{ij, mn} = \Braket{\phi_i^{R_m} | - \frac{1}{2}\nabla^2 | \phi_j^{R_n}}$, with
%
\begin{equation}
    N^E_{ij, mn} =
    \Braket{\phi_i^{R_m} | -\frac{1}{\lvert \symbf{r} - \symbf{R}_1 \rvert} | \phi_j^{R_n}}
    + \Braket{\phi_i^{R_m} | -\frac{1}{\lvert \symbf{r} - \symbf{R}_2 \rvert} | \phi_j^{R_n}},
\end{equation}
%
and
%
\begin{equation}
    N^N_{ij} = \Braket{\phi_i^{R_m} | \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert} | \phi_j^{R_n}}
\end{equation}
%
and $S_{ij, mn} = \Braket{\phi_i^{R_m} | \phi_j^{R_n}}$ are all $8 \times 8$ matrix,
where $i$,~$j=1$,~$2$,~$3$,~$4$, and $m, n=1$,~$2$.

The total energy is
%
\begin{equation}
    \begin{aligned}
        E_{\text{total}} & = 2E - \Braket{\psi | \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert} | \psi}
                         & = 2E - \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert},
    \end{aligned}
\end{equation}
%
here the second term is calculated twice in eigenvalue, so subtract it.

When $\lvert \symbf{R}_1 - \symbf{R}_2 \rvert = \qty{1.020}{\bohr}$, the eigenvalue is \qty{-1.88723984843}{\hartree}.
See Figure~\ref{fig:pro3a} for reference.

\subsection*{(b)}

Now the Hamiltonian becomes
%
\begin{multline}

    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_1 \rvert} - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_2 \rvert}\\
    - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_1 \rvert} - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_2 \rvert}

and

        + \int d\symbf{r}_2 \frac{\psi_2^\ast (\symbf{r}_2) \psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert}

Similar as $2(b)$, the eigenvalue problem now becomes
%
\begin{equation}

\subsection*{(c)}

The Hamiltonian is still the same, but Schrödinger equation will be
%
\begin{multline}\label{eq:psi1}
    \biggl(-\frac{1}{2}\nabla_1^2
    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_1 \rvert}
    - \frac{1}{\lvert \symbf{r}_1 - \symbf{R}_2 \rvert}
    + \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert}\\
    + \int d\symbf{r}_2
    \frac{\psi_2^\ast (\symbf{r}_2) \psi_2(\symbf{r}_2) - \psi_2^\ast (\symbf{r}_2) \hat{P} \psi_2(\symbf{r}_2)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert}
    \biggr)
    \psi_1(\symbf{r}_1) = E_1 \psi_1(\symbf{r}_1),
\end{multline}
%
and
%
\begin{multline}\label{eq:psi2}
    \biggl(-\frac{1}{2}\nabla_2^2
    - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_1 \rvert}
    - \frac{1}{\lvert \symbf{r}_2 - \symbf{R}_2 \rvert}
    + \frac{1}{\lvert \symbf{R}_1 - \symbf{R}_2 \rvert}\\
    + \int d\symbf{r}_1
    \frac{\psi_1^\ast (\symbf{r}_1) \psi_1(\symbf{r}_1) - \psi_1^\ast (\symbf{r}_1) \hat{P} \psi_1(\symbf{r}_1)}{\lvert \symbf{r}_1 - \symbf{r}_2 \rvert}
    \biggr)
    \psi_2(\symbf{r}_2) = E_2 \psi_2(\symbf{r}_2),
\end{multline}
%
where $\hat{P}$ is the permutation operator. So the eigenvalue problem will be
%
\begin{equation}
    \det(T_{ij, mn} + N^E_{ij, mn} + N^N_{ij, mn} + F_{ij, mn} - S_{ij, mn} E) = 0.
\end{equation}
%
where
%
\begin{multline}
    F_{ij, mn} = \sum_{\substack{p, q\\u, v}} 8\pi^2 c_{p, u}^\ast c_{q, v}
    \! \scaleint{7ex}_{\bs 0}^{\infty} r_1^2  \, dr_1
    \! \scaleint{7ex}_{\bs 0}^{\infty} r_2^2 \, dr_2
    \! \scaleint{7ex}_{\bs 0}^{\pi}
    \frac{\sin\theta\, d\theta}{\sqrt{r_1^2 + r_2^2 - 2 r_1 r_2 \cos\theta}}\\
    \cdot \Bigl(
    \bigl( \phi_i^{R_m} \bigr)^\ast(r_1) \bigl(\phi_p^{R_u} \bigr)^\ast(r_2) \phi_q^{R_v}(r_2)\phi_j^{R_n}(r_1)\\
    - \bigl( \phi_i^{R_m} \bigr)^\ast(r_1) \bigl(\phi_p^{R_u} \bigr)^\ast(r_2) \phi_j^{R_n}(r_2)\phi_q^{R_v}(r_1)
    \Bigr)
\end{multline}
%
Now perform self-consistency loop, first assume a trial $\psi_2$, substitute into
$\eqref{eq:psi1}$, solve the coefficients of $\psi_1$, then substitute this $\psi_1$
into $\eqref{eq:psi2}$, solve the new coefficients of $\psi_2$, etc. Repeat this procedure
until the total energy converges.
